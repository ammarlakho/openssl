        /* Test that a BIO using the cipher filter can encrypt and then decrypt
         * data correctly with AES‑256‑CBC. */
        const unsigned char key[32] = {
            0x00,0x01,0x02,0x03,0x04,0x05,0x06,0x07,
            0x08,0x09,0x0a,0x0b,0x0c,0x0d,0x0e,0x0f,
            0x10,0x11,0x12,0x13,0x14,0x15,0x16,0x17,
            0x18,0x19,0x1a,0x1b,0x1c,0x1d,0x1e,0x1f
        };
        const unsigned char iv[16] = {
            0x00,0x01,0x02,0x03,0x04,0x05,0x06,0x07,
            0x08,0x09,0x0a,0x0b,0x0c,0x0d,0x0e,0x0f
        };
        const unsigned char plaintext[] =
            "The quick brown fox jumps over the lazy dog";
        const size_t pt_len = sizeof(plaintext) - 1; /* exclude NUL */

        BIO *mem_enc = NULL, *enc = NULL;
        BIO *mem_dec = NULL, *dec = NULL;
        char *enc_data = NULL;
        long enc_len = 0;
        unsigned char outbuf[256];
        int out_total = 0, r;

        /* ---------- encrypt ---------- */
        mem_enc = BIO_new(BIO_s_mem());
        enc    = BIO_new(BIO_f_cipher());

        if (!TEST_ptr(mem_enc) || !TEST_ptr(enc))
            goto err;

        if (!TEST_int_gt(BIO_set_cipher(enc, EVP_aes_256_cbc(),
                                      key, iv, 1), 0))
            goto err;

        /* chain: enc -> mem_enc */
        BIO_push(enc, mem_enc);

        if (!TEST_int_eq(BIO_write(enc, plaintext, (int)pt_len),
                         (int)pt_len))
            goto err;

        if (!TEST_int_gt(BIO_flush(enc), 0))
            goto err;

        /* obtain encrypted bytes from the memory BIO */
        enc_len = BIO_get_mem_data(mem_enc, &enc_data);
        if (!TEST_int_gt(enc_len, 0))
            goto err;

        /* ---------- decrypt ---------- */
        mem_dec = BIO_new_mem_buf(enc_data, (int)enc_len);
        dec    = BIO_new(BIO_f_cipher());

        if (!TEST_ptr(mem_dec) || !TEST_ptr(dec))
            goto err;

        if (!TEST_int_gt(BIO_set_cipher(dec, EVP_aes_256_cbc(),
                                      key, iv, 0), 0))
            goto err;

        /* chain: dec -> mem_dec */
        BIO_push(dec, mem_dec);

        /* Read all decrypted data */
        out_total = 0;
        while ((r = BIO_read(dec,
                            outbuf + out_total,
                            (int)(sizeof(outbuf) - out_total))) > 0) {
            out_total += r;
            if (out_total == (int)sizeof(outbuf))
                break; /* should never happen for this test */
        }
        if (!TEST_int_eq(out_total, (int)pt_len))
            goto err;

        if (!TEST_mem_eq(outbuf, out_total, plaintext, pt_len))
            goto err;

        /* clean up */
        BIO_free_all(enc);
        BIO_free_all(dec);
        return 1;

    err:
        BIO_free_all(enc);
        BIO_free_all(dec);
        return 0;
        /* END_LLM_REPLACE */
